# Redirect all HTTP to HTTPS

server {
    listen 80;
    server_name api.rose-academy.com www.rose-academy.com rose-academy.com;
    return 301 https://$host$request_uri;
}

# ==============================
# Backend API (Laravel)
# ==============================
server {
    listen 443 ssl http2;
    server_name api.rose-academy.com;

    root /var/www/Laravel-Rose/public;
    index index.php index.html index.htm;

    ssl_certificate     /etc/ssl/certs/rose-api.pem;
    ssl_certificate_key /etc/ssl/private/rose-api.key;

    ssl_protocols TLSv1.2 TLSv1.3;
    ssl_ciphers HIGH:!aNULL:!MD5;
    client_max_body_size 2G;

    # ========================================
    # CORS setup for allowed origins
    # ========================================
    set $cors_origin "";
    if ($http_origin = "https://www.rose-academy.com") {
        set $cors_origin $http_origin;
    }
    if ($http_origin = "https://a38443b1-cb2e-499f-be8a-876f5feb90f7-00-kk7vg6v1qs5x.riker.replit.dev") {
        set $cors_origin $http_origin;
    }


    # ========================================
    # Static storage files
    # ========================================
    location /storage/ {
        alias /var/www/Laravel-Rose/storage/app/public/;
        add_header 'Access-Control-Allow-Origin' "$cors_origin" always;
        add_header 'Access-Control-Allow-Methods' 'GET, OPTIONS' always;
        add_header 'Access-Control-Allow-Headers' 'Authorization, Content-Type' always;
        add_header Cache-Control "public, max-age=31536000, immutable";
        try_files $uri =404;
    }

    # ========================================
    # Uploaded proofs (public)
    # ========================================
    location /public/ {
        alias /var/www/Laravel-Rose/public/uploads/payment_proofs/;
        add_header 'Access-Control-Allow-Origin' "$cors_origin" always;
        add_header 'Access-Control-Allow-Methods' 'GET, OPTIONS' always;
        add_header 'Access-Control-Allow-Headers' 'Authorization, Content-Type' always;
        add_header Cache-Control "public, max-age=31536000, immutable";
        try_files $uri =404;
    }

    # ========================================
    # Laravel app main logic
    # ========================================
    location / {
        # Handle preflight (OPTIONS) requests
        if ($request_method = OPTIONS) {
            add_header 'Access-Control-Allow-Origin' "$cors_origin" always;
            add_header 'Access-Control-Allow-Methods' 'GET, POST, PUT, PATCH, DELETE, OPTIONS' always;
            add_header 'Access-Control-Allow-Headers' 'Authorization, Content-Type, X-Requested-With, Accept, Accept-Language, Origin' always;
            add_header 'Access-Control-Allow-Credentials' 'true' always;
            add_header 'Access-Control-Max-Age' 86400;
            return 204;
        }

        try_files $uri $uri/ /index.php?$query_string;
    }

    # ========================================
    # PHP-FPM handling
    # ========================================
    location ~ \.php$ {
        include snippets/fastcgi-php.conf;
        fastcgi_pass unix:/var/run/php/php8.2-fpm.sock;
        fastcgi_param SCRIPT_FILENAME $realpath_root$fastcgi_script_name;
        include fastcgi_params;
    }

    # ========================================
    # Security rules
    # ========================================
    location ~ /\.(?!well-known).* {
        deny all;
    }
}

# ==============================
# Frontend (React Production Build)
# ==============================
server {
    listen 443 ssl http2;
    server_name www.rose-academy.com rose-academy.com;

    root /var/www/React-Rose/dist;
    index index.html;

    ssl_certificate     /etc/ssl/certs/rose-api.pem;
    ssl_certificate_key /etc/ssl/private/rose-api.key;

    ssl_protocols TLSv1.2 TLSv1.3;
    ssl_ciphers HIGH:!aNULL:!MD5;

    # ========================================
    # Cache static assets for performance
    # ========================================
    location ~* \.(?:ico|css|js|gif|jpe?g|png|woff2?|ttf|svg|eot)$ {
        expires 1y;
        add_header Cache-Control "public, max-age=31536000, immutable";
        try_files $uri /index.html;
    }

    # ========================================
    # SPA fallback for React router
    # ========================================
    location / {
        try_files $uri $uri/ /index.html;
    }

    # ========================================
    # Security
    # ========================================
    location ~ /\.(?!well-known).* {
        deny all;
    }
}
